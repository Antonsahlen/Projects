import java.util.ArrayList;

public class DogRegister {
    private static final int EXIT_COMMAND = 0;
    private static final int REGISTER_NEW_OWNER = 1;
    private static final int REGISTER_NEW_DOG = 2;
    private static final int REMOVE_OWNER = 3;
    private static final int LIST_OWNERS = 4;
    private static final int LIST_DOGS = 5;
    private static final int GIVE_DOG_TO_OWNER = 6;
    private static final int REMOVE_DOG = 7;
    private static final int REMOVE_DOG_FROM_OWNER = 8;
    private static final int INCREASE_AGE = 9;



    private InputReader inputReader = new InputReader();
    private DogCollection dogRegister = new DogCollection();
    private OwnerCollection ownerRegister = new OwnerCollection();


    private void start() {
        Initialize();
        runCommandoLoop();
        exitCommand();
    }

    private void exitCommand() {
        System.out.println("Exiting program");

    }

    private void runCommandoLoop() {
        int command;
        do {
            command = readCommand();
            if (command >= 0) {
                handleCommand(command);
            }

        } while (command != EXIT_COMMAND);
    }


    private void commandMenu() {
        System.out.println("Välj ett av alternativen 1-5");
        System.out.println("1. Registrera ny ägare");
        System.out.println("2. Registrera ny hund");
        System.out.println("3. Ta bort ägare");
        System.out.println("4. Lista ägare");
        System.out.println("5. Lista hund");
        System.out.println("6. Ge hund till ägare");
        System.out.println("7. Ta bort hund");
        System.out.println("0. Avsluta");
    }

    private void handleCommand(int command) {

        switch (command) {
            case REGISTER_NEW_OWNER:                               //Byta NAMN!!
                registerNewOwner();
                break;
            case REGISTER_NEW_DOG:
                registerNewDog();
                break;
            case REMOVE_OWNER:
                removeOwner();
                break;
            case LIST_OWNERS:
                listOwners();
                break;
            case LIST_DOGS:
                listDogs();
                break;
            case GIVE_DOG_TO_OWNER:
                giveDogToOwner();
                break;
            case REMOVE_DOG:
                removeDog();
                break;
            case EXIT_COMMAND:
                break;
            default:
                System.out.println("Fel kommando");

        }
    }


    //ÄNDRA!!!!!
    private void giveDogToOwner() {
        String dogName = inputReader.readLine("Vilken hund ska ges bort").trim();
        if (dogName.isEmpty()) {
            System.out.println("Fel, hundens namn saknas");
            return;
        }

        Dog dog = dogRegister.getDog(dogName);
        if (dog == null) {
            System.out.println("Fel, hunden finns inte");
            return;
        }

        String ownerName = inputReader.readLine("Vem ska bli ägare").trim();

        if (ownerName.isEmpty() || ownerName == null) {
            System.out.println("Fel, ägaren finns inte");
            return;
        }

        Owner owner = ownerRegister.getOwner(ownerName);
        if (owner == null) {
            System.out.println("Fel, ägaren finns inte");
            return;
        }
        if (dog.getOwner() != null){
            System.out.println("Fel, hunden har redan en ägare");
            return;
        }

        dog.setOwner(owner);

    }


    private void listDogs() {
        if (dogRegister.getDogs().isEmpty()) {
            System.out.println("Fel, inga hundar är registrerade");
            return;
        }

        double minTailLength = inputReader.readDouble("Ange minsta svanslängd");
        ArrayList<Dog> matchingDogs = dogRegister.getDogsMinTailLength(minTailLength);

        if (matchingDogs.isEmpty()) {
            System.out.println("Fel, inga hundar med tillräcklig svanslängd");
            return;
        }

        for (Dog dog : matchingDogs) {
            System.out.println(dog);
        }
    }


    private void registerNewDog() {
        String name;
        while (true) {
            name = inputReader.readLine("Hundens namn").trim();
            if (name.isEmpty()) {
                System.out.println("Fel");
                continue;
            }
            if (dogRegister.getDog(name) != null) {
                System.out.println("Fel");
                return;
            }
            break;
        }

        String breed;
        while (true) {
            breed = inputReader.readLine("Hundens ras").trim();
            if (breed.isEmpty()) {
                System.out.println("Fel");
                continue;
            }
            break;
        }

        int age = inputReader.readInt("Hundens ålder");
        int weight = inputReader.readInt("Hundens vikt");

        Dog dog = new Dog(name, breed, age, weight);
        dogRegister.addDog(dog);
        System.out.println(dog);
    }


    private void registerNewOwner() {
        while (true) {
            String name = inputReader.readLine("Ange ägarens namn").trim();

            if (name == null || name.isBlank()) {
                System.out.println("Fel");

            } else if (ownerRegister.containsOwner(name)) {
                System.out.println("Fel, ägaren finns redan");
                break;

            } else {
                Owner newOwner = new Owner(name);
                ownerRegister.addOwner(newOwner);
                System.out.println("Ny ägare registrerad: " + newOwner.getName());
                break;
            }

        }
    }

    private void removeOwner(){
        String nameToRemove = inputReader.readLine("Ange namnet på ägaren som ska tas bort: ");

        Owner ownerToRemove = ownerRegister.getOwner(nameToRemove);
        if (ownerToRemove == null) {
            System.out.println("Fel");
            return;
        }
        for (Dog dog : ownerToRemove.getDogs()) {
            dog.setOwner(null);
        }
        ownerRegister.removeOwner(ownerToRemove);
    }

    private void removeDog() {

            String name = inputReader.readLine("Ange namnet på hunden som ska tas bort ");
            Dog dog = dogRegister.getDog(name);

            if (dog == null) {
                System.out.println("Fel: Hunden finns inte.");
                return;
            }

            if (dog.getOwner() != null) {
                System.out.println("Fel: Hunden kan inte tas bort eftersom den har en ägare.");
                return;
            }

            boolean success = dogRegister.removeDog(name);
            if (!success) {
                System.out.println("Fel: Kunde inte ta bort hunden.");
            }
        }


    private void listOwners() {
        ArrayList<Owner> owners = ownerRegister.getOwners();

        if (owners.isEmpty()) {
            System.out.println("Fel, inga ägare är registrerade");
            return;

        }
            System.out.println("Registrerade ägare: ");
            for (Owner owner : owners) {
                System.out.println(owner.getName());

                ArrayList<Dog> dogs = owner.getDogs();
                for (Dog dog : dogs) {
                    System.out.println(dog.getName());
                }
            }

        }



    private int readCommand() {

        String inputLine = inputReader.readLine("Enter command").trim();

        if (inputLine.equalsIgnoreCase("exit")) {
            return EXIT_COMMAND;
        }
        if (inputLine.equalsIgnoreCase("REGISTER NEW OWNER")) {
            return REGISTER_NEW_OWNER;
        }
        if (inputLine.equalsIgnoreCase("REGISTER NEW DOG")) {
            return REGISTER_NEW_DOG;
        }
        if (inputLine.equalsIgnoreCase("LIST OWNERS")) {
            return LIST_OWNERS;
        }
        if (inputLine.equalsIgnoreCase("REMOVE OWNER")) {
            return REMOVE_OWNER;
        }
        if (inputLine.equalsIgnoreCase("LIST DOGS")) {
            return LIST_DOGS;
        }
        if(inputLine.equalsIgnoreCase("GIVE DOG TO OWNER")){
            return GIVE_DOG_TO_OWNER;
        }
        if (inputLine.toUpperCase().startsWith("GIVE ")) {
            return GIVE_DOG_TO_OWNER;
        }
        if(inputLine.equalsIgnoreCase("REMOVE DOG")){
            return REMOVE_DOG;
        }

        try {
            int command = Integer.parseInt(inputLine);
            if (command >= 0 && command <= 4) {
                return command;
            } else {
                System.out.println("Fel kommando, försök igen");
                return -1;
            }
        } catch (NumberFormatException e) {
            System.out.println("Fel kommando, försök igen");
            return -1;
        }

    }

    private void Initialize() {
        System.out.println("Initializing program");

    }

    public static void main(String[] args) {
        new DogRegister().start();
    }
}
