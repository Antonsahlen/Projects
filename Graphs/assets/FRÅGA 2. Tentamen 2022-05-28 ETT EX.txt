public abstract class MusicCollection implements Music(){

public abstract List<Song> getSongs();

public List <String> getArtist(){
	List <String> list = new ArrayList<>();
	for ( Song song : getSongs()){
		String artist = song.getArtist();
		list.add(artist);
}

	return list;

}

@Override 
public int getPlaytime(){
	int sum = 0;
	for (Song song : song.getSongs()){
		int playtime = song.getPlaytime();
		sum += playtime
		

} 
	return sum;

}
