    private int readCommand() {

        String inputLine = inputReader.readLine("Enter command").trim();

        if (inputLine.equalsIgnoreCase("exit")) {
            return EXIT_COMMAND;
        }
        if (inputLine.equalsIgnoreCase("REGISTER NEW OWNER")) {
            return REGISTER_NEW_OWNER;
        }
        if (inputLine.equalsIgnoreCase("REGISTER NEW DOG")) {
            return REGISTER_NEW_DOG;
        }
        if (inputLine.equalsIgnoreCase("LIST OWNERS")) {
            return LIST_OWNERS;
        }
        if (inputLine.equalsIgnoreCase("REMOVE OWNER")) {
            return REMOVE_OWNER;
        }
        if (inputLine.equalsIgnoreCase("LIST DOGS")) {
            return LIST_DOGS;
        }
        if(inputLine.equalsIgnoreCase("GIVE DOG TO OWNER")){
            return GIVE_DOG_TO_OWNER;
        }
        if (inputLine.toUpperCase().startsWith("GIVE ")) {              //Behövs denna?
            return GIVE_DOG_TO_OWNER;
        }
        if(inputLine.equalsIgnoreCase("REMOVE DOG")){
            return REMOVE_DOG;
        }
        if(inputLine.equalsIgnoreCase("REMOVE DOG FROM OWNER")){
            return REMOVE_DOG_FROM_OWNER;
        }
        if(inputLine.equalsIgnoreCase("INCREASE AGE")){
            return INCREASE_AGE;
        }

        try {
            int command = Integer.parseInt(inputLine);
            if (command >= 0 && command <= 4) {
                return command;
            } else {
                System.out.println("Fel kommando, försök igen");
                return -1;
            }
        } catch (NumberFormatException e) {
            System.out.println("Fel kommando, försök igen");
            return -1;
        }

    }
